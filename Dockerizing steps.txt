1. Create .dockerignore (Inside the backend folder and paste:

node_modules
npm-debug.log
.env
.git
.vscodes



2. Create Dockerfile (Inside the backend folder) and paste:

# Use an official Node.js runtime as the base image
FROM node:20-alpine

# Set the working directory inside the container
WORKDIR /app

# Copy package.json and package-lock.json (if available) first to cache dependencies
COPY package*.json ./

# Install dependencies inside the container
RUN npm install

# Copy the rest of the backend application code
COPY . .

# Expose the port your JS backend runs on (replace 5000 with your actual JS backend port if different)
EXPOSE 5000

# Command to run your development script
CMD ["npm", "run", "dev"]


3. Create .dockerignore (Inside the frontend folder) and paste:

node_modules
dist
.git
.vscode
.env


4. Create Dockerfile (Inside the frontend folder) and paste:

# Use an official Node.js runtime as the base image
FROM node:20-alpine

# Set the working directory inside the container
WORKDIR /app

# Copy package configuration files first to optimize layer caching
COPY package*.json ./

# Install project dependencies
RUN npm install

# Copy the rest of the frontend source code
COPY . .

# Expose the Vite development server port (Vite typically defaults to 5173)
EXPOSE 5173

# Vite requires host binding (0.0.0.0) to be accessible outside the container
# If your package.json script is just "vite", you can append -- --host or configure it here
CMD ["npm", "run", "dev", "--", "--host", "0.0.0.0"]




5. update docker-compose.yml:

version: '3.8'

services:
  # Local PostgreSQL Service
  postgres-db:
    image: postgres:17-alpine
    container_name: typ_postgres_db
    restart: always
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: my_local_password
      POSTGRES_DB: kutlwanong_lms_db
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql

  # .NET Core API Service
  backend-api:
    build:
      context: ./backend/dotnetAPI-ver-2/thirdYearProject
      dockerfile: Dockerfile
    container_name: typ_backend_api
    restart: always
    ports:
      - "7060:7060"
    environment:
      - ASPNETCORE_ENVIRONMENT=Development
      - ConnectionStrings__DefaultConnection=Host=postgres-db;Port=5432;Database=kutlwanong_lms_db;Username=postgres;Password=my_local_password;
    depends_on:
      - postgres-db
    volumes:
      - typ_uploads_data:/app/Uploads

  # JavaScript Backend Service (Node.js)
  backend-js:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: typ_backend_js
    restart: always
    ports:
      - "5000:5000" # Change 5000 to whatever port your JS backend runs on
    environment:
      - NODE_ENV=development
      # Use container name 'postgres-db' instead of localhost
      - DB_HOST=postgres-db
      - DB_PORT=5432
      - DB_USER=postgres
      - DB_PASSWORD=my_local_password
      - DB_NAME=kutlwanong_lms_db
    depends_on:
      - postgres-db
    volumes:
      - ./backend:/app
      - /app/node_modules # Prevents local node_modules from overwriting container's installed modules

  # Frontend Service (React / Vite)
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    container_name: typ_frontend
    restart: always
    ports:
      - "5173:5173"
    environment:
      - VITE_DOTNET_API_URL=http://localhost:7060
      - VITE_JS_API_URL=http://localhost:5000
    depends_on:
      - backend-api
      - backend-js
    volumes:
      - ./frontend:/app
      - /app/node_modules # Prevents local node_modules from overwriting container's modules

# Persistent volumes for files and database
volumes:
  typ_uploads_data:
  postgres_data:




6. in base folder run docker compose up --build

7. use docker compose down to stop everything

docker compose down -v
docker compose build --no-cache
